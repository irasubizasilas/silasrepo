import java.util.Scanner;

 

public class BicycleRental {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

 

        // 1. Read inputs

        System.out.print("Enter starting hour (0-23): ");

        int startTime = input.nextInt();

 

        System.out.print("Enter ending hour (1-24): ");

        int endTime = input.nextInt();

 

        // 2. Validate input constraints

        if (startTime < 0 || startTime > 23 || endTime < 1 || endTime > 24) {

            System.out.println("Error: Start hour must be 0-23 and end hour must be 1-24.");

            return;

        }

        if (startTime >= endTime) {

            System.out.println("Error: Starting time must be strictly less than ending time.");

            return;

        }

 

        int totalRent = 0;

 

        // 3. Loop through each rented hour and apply rates

        for (int hour = startTime; hour < endTime; hour++) {

            if ((hour >= 0 && hour < 7) || (hour >= 21 && hour < 24)) {

                totalRent += 500;  // 500 RWF bracket

            } else if ((hour >= 7 && hour < 14) || (hour >= 19 && hour < 21)) {

                totalRent += 1000; // 1000 RWF bracket

            } else if (hour >= 14 && hour < 19) {

                totalRent += 1500; // 1500 RWF bracket

            }

        }

 

        // 4. Output the results

        System.out.println("Total rental cost: " + totalRent + " RWF");

        input.close();

    }

}

import java.util.Scanner;

 

public class MushroomIdentifier {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);

       

        System.out.println("--- Mushroom Identifier ---");

        System.out.println("Think of one of these: Agaric Jaunissant, Amanite tue-mouche, Cepe de bordeau, Coprin chevelu, Pied bleu.\n");

 

        // Question 1: Gills vs Pores (Splits Cepe de Bordeaux from the rest immediately)

        System.out.print("Does your mushroom have gills? (yes/no): ");

        String gills = input.next().trim().toLowerCase();

 

        if (gills.equals("no")) {

            // Cepe de Bordeaux is the only one with pores (no gills)

            System.out.println("\nYour mushroom is: Cepe de bordeaux");

        } else {

            // Question 2: Habitat (Splits Meadow dwellers from Forest dwellers)

            System.out.print("Does your mushroom grow in a forest? (yes/no): ");

            String forest = input.next().trim().toLowerCase();

 

            if (forest.equals("no")) {

                // Grows in a meadow: Agaric Jaunissant vs Coprin chevelu

                // Question 3: Convex Cup splits these two

                System.out.print("Does your mushroom have a convex cup? (yes/no): ");

                String convexCup = input.next().trim().toLowerCase();

 

                if (convexCup.equals("yes")) {

                    System.out.println("\nYour mushroom is: Agaric Jaunissant");

                } else {

                    System.out.println("\nYour mushroom is: Coprin chevelu");

                }

 

            } else {

                // Grows in a forest: Amanite tue-mouche vs Pied bleu

                // Question 3: Ring splits these two

                System.out.print("Does your mushroom have a ring? (yes/no): ");

                String ring = input.next().trim().toLowerCase();

 

                if (ring.equals("yes")) {

                    System.out.println("\nYour mushroom is: Amanite tue-mouche");

                } else {

                    System.out.println("\nYour mushroom is: Pied bleu");

                }

            }

        }

       

        input.close();

    }

}
